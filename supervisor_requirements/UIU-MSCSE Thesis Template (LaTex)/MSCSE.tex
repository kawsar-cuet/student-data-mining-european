\chapter*{Notes/Instructions: For Writing the Appendix Chapter}
\addcontentsline{toc}{chapter}{Notes/Instructions: For Writing the Appendix Chapter}
\label{appendix:tips}

\begin{itemize}
  \item \textbf{Reference clearly in the text:} Always refer to the appendix in the main chapters (e.g., ``See Appendix C for the architecture details.'').
  \item \textbf{Label consistently:} Use Appendix A, B, C, etc., with subsections like A.1, A.2 for clarity and organization.
  \item \textbf{Follow formatting conventions:} Maintain the same font, layout, and citation styles used in the main body.
  \item \textbf{Avoid redundancy:} Do not repeat essential data or results already presented in the main text.
  \item \textbf{Avoid new findings:} Appendices should support—not introduce—your arguments or conclusions.
  \item \textbf{Use external links wisely:} When dealing with large codebases or datasets, provide GitHub links or data repositories rather than including raw data directly.
  \item \textbf{Examples:} There are some examples of what you can add in the appendix section. You can get the idea from them, but you should consult with your supervisor on this.
\end{itemize}

\section*{Gradient of the Cost Function}
\begin{align*}
J(\theta) &= \frac{1}{2m} \sum_{i=1}^{m} (h_\theta(x^{(i)}) - y^{(i)})^2 \\
\frac{\partial J(\theta)}{\partial \theta_j} &= \frac{1}{m} \sum_{i=1}^{m} (h_\theta(x^{(i)}) - y^{(i)}) x_j^{(i)}
\end{align*}
